# htp_project

Code for HTP project 

run it like this 

python path/sc2l_main.py `
 --encoder MLP `
 --dataset_name taylor2020 `
 --data_dir "path/SCLSC-code/" `
 --model_dir "path/SCLSC-code/models" `
 --gene_set hvg `
 --margin 1.0 `
 --output_dir "path/SCLSC-code/output" `
 --epoch 100 `
 --batch_size 256
